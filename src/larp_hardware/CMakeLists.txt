cmake_minimum_required(VERSION 3.22)
project(larp_hardware LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(depthai 3.10 QUIET CONFIG)
if(depthai_FOUND)
 set(OAK_SDK_TARGET depthai::core)
else()
 find_package(depthai_v3 CONFIG REQUIRED)
 set(OAK_SDK_TARGET depthai_v3::core)
endif()
find_package(yaml-cpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(flight_executive_msgs REQUIRED)
add_executable(px4_bench_node src/px4_bench_node.cpp)
target_include_directories(px4_bench_node PRIVATE include)
ament_target_dependencies(px4_bench_node rclcpp nav_msgs std_msgs px4_msgs flight_executive_msgs)
target_link_libraries(px4_bench_node Eigen3::Eigen)
install(TARGETS px4_bench_node DESTINATION lib/${PROJECT_NAME})
add_executable(oak_native_node src/oak_native_node.cpp)
target_include_directories(oak_native_node PRIVATE include)
ament_target_dependencies(oak_native_node rclcpp sensor_msgs std_msgs geometry_msgs)
target_link_libraries(oak_native_node ${OAK_SDK_TARGET} yaml-cpp)
install(TARGETS oak_native_node DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
 add_executable(frame_contract_test test/frame_contract_test.cpp)
 target_include_directories(frame_contract_test PRIVATE include ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/ov_core/src)
 target_link_libraries(frame_contract_test Eigen3::Eigen)
 target_compile_options(frame_contract_test PRIVATE -UNDEBUG)
 add_test(NAME frame_contract COMMAND frame_contract_test)
 add_executable(command_gate_test test/command_gate_test.cpp)
 target_include_directories(command_gate_test PRIVATE include)
 target_compile_options(command_gate_test PRIVATE -UNDEBUG)
 add_test(NAME command_gate COMMAND command_gate_test)
 add_executable(imu_alignment_test test/imu_alignment_test.cpp)
 target_include_directories(imu_alignment_test PRIVATE include)
 # Tests remain active in Release builds.
 target_compile_options(imu_alignment_test PRIVATE -UNDEBUG)
 add_test(NAME experimental_imu_alignment COMMAND imu_alignment_test)
endif()
ament_package()
