project(larp_navigation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)

find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(PkgConfig REQUIRED)
# Ubuntu 22.04 (Jetson/Humble) installs version-suffixed PCL pkg-config files.
pkg_search_module(PCL_COMMON REQUIRED IMPORTED_TARGET pcl_common pcl_common-1.12)
pkg_search_module(PCL_IO REQUIRED IMPORTED_TARGET pcl_io pcl_io-1.12)
pkg_search_module(PCL_FILTERS REQUIRED IMPORTED_TARGET pcl_filters pcl_filters-1.12)
set(SUPER_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/SUPER)
file(GLOB_RECURSE ROG_SRC CONFIGURE_DEPENDS
  ${SUPER_ROOT}/rog_map/src/rog_map/*.cpp
  ${SUPER_ROOT}/rog_map/include/*.cpp)
add_library(rog_map_core STATIC ${ROG_SRC})
target_include_directories(rog_map_core PUBLIC ${SUPER_ROOT}/rog_map/include ${PCL_CONVERSIONS_INCLUDE})
target_compile_definitions(rog_map_core PUBLIC USE_ROS2 ORIGIN_AT_CORNER FMT_HEADER_ONLY
  ROOT_DIR="${CMAKE_CURRENT_BINARY_DIR}/")
target_link_libraries(rog_map_core Eigen3::Eigen PkgConfig::PCL_COMMON PkgConfig::PCL_IO PkgConfig::PCL_FILTERS yaml-cpp)
ament_target_dependencies(rog_map_core rclcpp sensor_msgs visualization_msgs pcl_msgs message_filters)
file(GLOB_RECURSE SUPER_SRC CONFIGURE_DEPENDS
  ${SUPER_ROOT}/super_planner/src/*.cpp)
# The upstream demonstration FSM does not implement the PX4 mission contract.
list(FILTER SUPER_SRC EXCLUDE REGEX "/fsm.cpp$")
add_library(super_core STATIC ${SUPER_SRC})
# Use libc stack symbols without requiring optional libdw development headers.
target_compile_definitions(super_core PUBLIC BACKWARD_HAS_BACKTRACE_SYMBOL=1)
target_include_directories(super_core PUBLIC ${SUPER_ROOT}/super_planner/include)
target_link_libraries(super_core rog_map_core)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/log)
include(CTest)
add_library(navigation_interfaces INTERFACE)
target_include_directories(navigation_interfaces INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/navigation/include)
target_link_libraries(navigation_interfaces INTERFACE Eigen3::Eigen)
target_link_libraries(super_core navigation_interfaces)
add_executable(body_odometry_node src/navigation/body_odometry_node.cpp)
target_link_libraries(body_odometry_node navigation_interfaces yaml-cpp)
ament_target_dependencies(body_odometry_node rclcpp nav_msgs std_msgs geometry_msgs)
add_executable(ev_source_node src/navigation/ev_source_node.cpp)
target_link_libraries(ev_source_node navigation_interfaces)
ament_target_dependencies(ev_source_node rclcpp nav_msgs std_msgs)
add_executable(depth_map_node src/navigation/depth_map_node.cpp)
target_link_libraries(depth_map_node super_core navigation_interfaces)
ament_target_dependencies(depth_map_node rclcpp sensor_msgs nav_msgs std_msgs dronehacks_msgs)
